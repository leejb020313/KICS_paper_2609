\begin{document}

\begin{tikzpicture}[

    box/.style={draw=ink, rounded corners=1.5pt, align=center, inner sep=2pt, minimum height=7.5mm, line width=0.5pt},

    ours/.style={box, draw=ours, fill=ours!10, line width=0.8pt},

  \draw[flow] (svm) -- (route);

  \draw[flow] (route) -- node[lbl, right] {예} (llm);

  \draw[flow] (svm.south) -- node[lbl, right, pos=0.4] {$d$} (svm.south |- fuse.north);
